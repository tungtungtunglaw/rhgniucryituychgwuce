        item.Badge.Text = "[" .. Config[configKey].Name .. "]"

            item.Label.TextColor3 = active and Color3_fromRGB(100, 255, 140) or Color3_fromRGB(255, 100, 100)

end

local function addHistory(category, message)
    local timestamp = os_date("%H:%M:%S")
    local entry = { Time = timestamp, Category = category, Message = message }
    
    -- Inserts new notification at top of list
    table_insert(HistoryLogs, 1, entry)
    
    -- Enforces 100 limit per page; removes oldest when exceeded
    if #HistoryLogs > 100 then
        table_remove(HistoryLogs, #HistoryLogs)
    end
    
    if updateHistoryUi then 
        updateHistoryUi() 
    end
end

local function switchTab(name)
    for tabName, btn in pairs(TabButtons) do
        if tabName == name then
            btn.BackgroundColor3 = Config.RgbColor
            btn.TextColor3 = Color3_fromRGB(255, 255, 255)
        else
            btn.BackgroundColor3 = C_PANEL
            btn.TextColor3 = C_TEXT_DIM
        end
    end
    for tabName, page in pairs(TabPages) do
        page.Visible = (tabName == name)
    end
end

local function moveRoot(root, position)
    root.AssemblyLinearVelocity = Vector3_zero
    root.AssemblyAngularVelocity = Vector3_zero
    root.CFrame = position
end

local function isKnocked(character)
    local humanoid = FindFirstChildOfClass(character, "Humanoid")
    if not humanoid or humanoid.Health <= 0 then return true end

    local bodyEffects = FindFirstChild(character, "BodyEffects")
    for i = 1, 3 do
        local name = STATE_NAMES[i]
        local state = FindFirstChild(character, name)
        if GetAttribute(character, name) == true or (state and IsA(state, "BoolValue") and state.Value) then
            return true
        end
        if bodyEffects then
            state = FindFirstChild(bodyEffects, name)
            if GetAttribute(bodyEffects, name) == true or (state and IsA(state, "BoolValue") and state.Value) then
                return true
            end
        end
    end
    return false
end

function restoreState()
    local character = LocalPlayer.Character
    local root = character and FindFirstChild(character, "HumanoidRootPart")
    if character == savedCharacter and root and IsA(root, "BasePart") and savedPosition then
        moveRoot(root, savedPosition)
    end
    if changedFallHeight then
        workspace.FallHeightEnabled = oldFallHeight
        changedFallHeight = false
    end
    savedCharacter = nil
    savedPosition = nil
end

function enableAntiVoid()
    if not changedFallHeight then
        oldFallHeight = workspace.FallHeightEnabled
        workspace.FallHeightEnabled = false
        changedFallHeight = true
    end
end

function executeKickAction()
    if #TargetUsers == 0 then
        notify("Kick Failed", "No players currently selected in target list.")
        return
    end

    local kickCount = 0
    for _, tData in ipairs(TargetUsers) do
        local targetChar = FindFirstChild(WorkspaceCharacters, tData.Name)
        if targetChar then
            local specialParts = FindFirstChild(targetChar, "SpecialParts")
            local targetRoot = specialParts and FindFirstChild(specialParts, "HumanoidRootPart") or FindFirstChild(targetChar, "HumanoidRootPart")
            if targetRoot and IsA(targetRoot, "BasePart") then
                FireServer(MainEvent, "RopeCharacter", targetRoot)
                kickCount = kickCount + 1
            end
        end
    end

    addHistory("Kick", "Executed server kick action on " .. kickCount .. " targets.")
    notify("Kick Executed", "Target kick event sent for " .. kickCount .. " players.")
end

-- Main Simulation Loop
updateConnection = RunService.PreSimulation:Connect(function()
    if (not VoidAllActive and not VoidTargetActive and not CustomYActive) or not SystemActive then return end

    local character = LocalPlayer.Character
    local humanoid = character and FindFirstChildOfClass(character, "Humanoid")
    local root = character and FindFirstChild(character, "HumanoidRootPart")

    if not humanoid or humanoid.Health <= 0 or not root or not IsA(root, "BasePart") then return end

    if savedCharacter ~= character then
        savedCharacter = character
        savedPosition = root.CFrame
        nextEquip = 0
        ropeCycleStart = os_clock()
        isCyclePausing = false
    end

    local rootCFrame = root.CFrame
    moveRoot(root, CFrame_new(rootCFrame.X, HoldYPosition, rootCFrame.Z))

    if CustomYActive then return end

    local now = os_clock()
    local cycleElapsed = now - ropeCycleStart

    if not isCyclePausing then
        if cycleElapsed >= ACTIVE_ROPE_TIME then
            isCyclePausing = true
            ropeCycleStart = now
            FireServer(MainEvent, "RopeCharacter", nil)
            addHistory("Tether", "Rope tether reset cycle executed.")
            return
        end
    else
        if cycleElapsed >= PAUSE_ROPE_TIME then
            isCyclePausing = false
            ropeCycleStart = now
        else
            return
        end
    end

    local rope = FindFirstChild(character, TOOL_NAME)
    if not rope or not IsA(rope, "Tool") then
        if now < nextEquip then return end
        nextEquip = now + EQUIP_INTERVAL
        local backpack = FindFirstChildOfClass(LocalPlayer, "Backpack")
        rope = backpack and FindFirstChild(backpack, TOOL_NAME)
        if not rope or not IsA(rope, "Tool") then return end
        humanoid:EquipTool(rope)
        addHistory("Tether", "Auto-equipped tool: " .. TOOL_NAME)
        if rope.Parent ~= character then return end
    end

    if now < nextUpdate then return end
    nextUpdate = now + UPDATE_INTERVAL

    if VoidAllActive then
        local chars = WorkspaceCharacters:GetChildren()
        for i = 1, #chars do
            local target = chars[i]
            if target ~= character and not isKnocked(target) then
                local specialParts = FindFirstChild(target, "SpecialParts")
                local targetRoot = specialParts and FindFirstChild(specialParts, "HumanoidRootPart")
                if targetRoot and IsA(targetRoot, "BasePart") then
                    FireServer(MainEvent, "RopeCharacter", targetRoot)
                end
            end
        end
    elseif VoidTargetActive and #TargetUsers > 0 then
        for i = 1, #TargetUsers do
            local tName = TargetUsers[i].Name
            local target = FindFirstChild(WorkspaceCharacters, tName)
            if target and target ~= character and not isKnocked(target) then
                local specialParts = FindFirstChild(target, "SpecialParts")
                local targetRoot = specialParts and FindFirstChild(specialParts, "HumanoidRootPart")
                if targetRoot and IsA(targetRoot, "BasePart") then
                    FireServer(MainEvent, "RopeCharacter", targetRoot)
                end
            end
        end
    end
end)

local function getClosestPlayerToMouse()
    local mouseLocation = UserInputService:GetMouseLocation()
    local closestPlayer = nil
    local shortestDistance = Config.MaxDistance
    local allPlayers = Players:GetPlayers()

    for i = 1, #allPlayers do
        local player = allPlayers[i]
        if player ~= LocalPlayer and player.Character then
            local rootPart = FindFirstChild(player.Character, "HumanoidRootPart")
            if rootPart then
                local screenPosition, onScreen = Camera:WorldToViewportPoint(rootPart.Position)
                if onScreen then
                    local distance = (Vector2_new(screenPosition.X, screenPosition.Y) - mouseLocation).Magnitude
                    if distance < shortestDistance then
                        shortestDistance = distance
                        closestPlayer = player
                    end
                end
            end
        end
    end
    return closestPlayer
end

local function triggerKeyVisual(keyCode, isPressed)
    for configKey, item in pairs(KeybindHudItems) do
        if Config[configKey] == keyCode then
            if isPressed then
                item.Frame.BackgroundColor3 = Color3_fromRGB(50, 65, 100)
                item.Badge.TextColor3 = Color3_fromRGB(255, 255, 255)
            else
                item.Frame.BackgroundColor3 = Color3_fromRGB(20, 22, 32)
                item.Badge.TextColor3 = Color3_fromRGB(255, 220, 100)
            end
        end
    end
end

UserInputService.InputBegan:Connect(function(input, gameProcessed)
    if currentlyRebinding then
        if input.UserInputType == Enum.UserInputType.Keyboard and input.KeyCode ~= Enum.KeyCode.Unknown then
            Config[currentlyRebinding] = input.KeyCode
            addHistory("System", "Rebound " .. currentlyRebinding .. " to [" .. input.KeyCode.Name .. "]")
            currentlyRebinding = nil
            updateHudKeybinds()
            switchTab("Config")
        end
        return
    end

    if gameProcessed then return end
    local k = input.KeyCode

    triggerKeyVisual(k, true)

    if k == Config.HideGuiKeybind then
        MainFrame.Visible = not MainFrame.Visible
        return
    end

    if k == Config.ToggleKeybind then
        SystemActive = not SystemActive
        addHistory("System", "Engine toggled " .. (SystemActive and "ON" or "OFF"))
        notify("Engine State", "Engine system " .. (SystemActive and "ENABLED" or "DISABLED"))
        if not SystemActive then
            stopVoidAll()
            stopVoidTarget()
            stopCustomY()
        end
        updateHudKeybinds()
        return
    end

    if k == Config.ExecuteKeybind then
        if not SystemActive then return end
        executeKickAction()
        return
    end

    if k == Config.GrabKeybind then
        if not SystemActive then return end
        local targetPlayer = getClosestPlayerToMouse()
        if targetPlayer then
            local exists = false
            for _, tData in ipairs(TargetUsers) do
                if tData.Name == targetPlayer.Name then exists = true break end
            end
            if not exists then
                table_insert(TargetUsers, { Name = targetPlayer.Name, Status = "Active" })
                addHistory("Target", "Added target via hotkey: " .. targetPlayer.Name)
                rebuildTargetList()
                rebuildMainTargetList()
                notify("Target Added", targetPlayer.Name .. " added to active list.")
            end
        end
        return
    end

    if k == Config.RemoveActiveKeybind then
        if not SystemActive then return end
        TargetUsers = {}
        stopVoidTarget()
        rebuildTargetList()
        rebuildMainTargetList()
        addHistory("Target", "Cleared all active targets via hotkey.")
        notify("Targets Cleared", "Target list reset.")
        return
    end
end)

UserInputService.InputEnded:Connect(function(input, gameProcessed)
    if input.UserInputType == Enum.UserInputType.Keyboard then
        triggerKeyVisual(input.KeyCode, false)
    end
end)

-- Main Frame Dragging Implementation
local dragging, dragStart, startPos
Header.InputBegan:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseButton1 then
        dragging = true
        dragStart = input.Position
        startPos = MainScaleFrame.Position
        input.Changed:Connect(function()
            if input.UserInputState == Enum.UserInputState.End then dragging = false end
        end)
    end
end)

UserInputService.InputChanged:Connect(function(input)
    if dragging and input.UserInputType == Enum.UserInputType.MouseMovement then
        local delta = input.Position - dragStart
        MainScaleFrame.Position = UDim2.new(
            startPos.X.Scale, startPos.X.Offset + delta.X,
            startPos.Y.Scale, startPos.Y.Offset + delta.Y
        )
    end
    if hudDragging and input.UserInputType == Enum.UserInputType.MouseMovement then
        local delta = input.Position - hudDragStart
        HudScaleFrame.Position = UDim2.new(
            hudStartPos.X.Scale, hudStartPos.X.Offset + delta.X,
            hudStartPos.Y.Scale, hudStartPos.Y.Offset + delta.Y
        )
    end
end)

    MainFrame.Visible = false
    notify("System", "Engine UI Hidden. Press [" .. Config.HideGuiKeybind.Name .. "] to show.")
end)

    if updateConnection then updateConnection:Disconnect() end
    stopVoidAll()
    stopVoidTarget()
    stopCustomY()
end)
